    ```typescript
    export class DinoRankApiClient {
      public email: string
      public pass: string
      private cookies: string = ''
      private sessionFile: string

      constructor(email: string, pass: string, sessionFile?: string)
      async login(language?: string): Promise<'ok' | 'device_conflict' | 'failed'>
      async get(url: string, referer?: string): Promise<string>
      async post(url: string, body: string, referer: string, maxRetries?: number): Promise<string>
      async logout(): Promise<void>
      extractContentCredits(html: string): number
      getCookieHeader(): string
    }
    ```

    **login(language?) implementation:**
    1. Determine URLs based on language: `isEn = language === 'en'`
       - loginUrl = `https://dinorank.com/${isEn ? 'en/' : ''}login/`
       - homedUrl = `https://dinorank.com/${isEn ? 'en/' : ''}homed/`
    2. GET loginUrl to obtain PHPSESSID + CSRF cookies
    3. POST `https://dinorank.com/ajax/login.php` with body:
       `nombreUsuario=${email}&clave=${pass}&permanecer=no&elemento=&tiempo=${Date.now()}`
       **CRITICAL: use `permanecer=no` (NOT `permanecer=si` like scrape-dinorank.ts)**
    4. Check response: if contains `status":"activo"` → GET homedUrl → save session → return 'ok'
    5. If contains 'dispositivo' or 'device' or 'otro' → return 'device_conflict'
    6. Otherwise → return 'failed'

    **post(url, body, referer, maxRetries=3) implementation:**
    - Wrap fetch in try/catch. On ECONNRESET / EPIPE / socket errors, retry up to maxRetries with 1s delay.
    - Merge cookies from response.

    **extractContentCredits(html) implementation — 4 regex patterns:**
    ```typescript
    const patterns = [
      /DinoBRAIN[:\s]*(\d+)/i,
      /Content[:\s]*(\d+)/i,
      /Contenido[:\s]*(\d+)/i,
      /créditos?\s+de\s+contenido[:\s]*(\d+)/i,
    ]
    ```
    Return first match as number, or 0 if none.

    - Default path: `content/dinorank-kw-session.json`

    - On logout: read file, delete `sessions[email]`, write file (keep other accounts)
    - On constructor: if session file exists and has entry for this email, restore cookies
