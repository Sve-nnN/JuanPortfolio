    ```typescript
    export interface DinoBrainParams {
      keyword: string

      siteType?: string     // 'nicho'|'negociolocal'|'ecommerce'|'producto', default 'nicho'
      domain?: string       // e.g. 'juan-tech.com'
      numWords?: number     // default 2000
      context?: string      // additional writing context
      exclusions?: string   // content exclusions

    }

    export interface DinoBrainResult {
      title: string
      markdown: string
      accountEmail: string
    }
    ```

    **Exported class:**
    ```typescript
    export class DinoBrainApiAdapter {
      async generate(params: DinoBrainParams): Promise<DinoBrainResult>
    }
    ```

    **generate() implementation — follow RESEARCH.md flow exactly:**

    1. `loadRegistry()` → filter `contentCredits > 0` and not expired
    2. Filter out accounts with active cooldown (`cooldownUntil` as ISO string > `new Date().toISOString()`)
    3. Score accounts for profile match:
       - `+10` if `createdLanguage === params.language`
       - `+6` if `createdCountry === params.country`
       - `+4` if `createdDomain === params.domain`
       - `+3` if `createdProjectType === params.siteType`
    4. Sort: score desc → contentCredits desc → lastUsed asc
    5. If no accounts available, throw `Error('No accounts with content credits available')`

    6. Try login with sorted accounts (skip to next on 'device_conflict' or 'failed'):
       ```typescript
       const api = new DinoRankApiClient(account.email, account.password)
       const loginResult = await api.login(language)
       if (loginResult !== 'ok') continue // try next account
       ```

    7. In try/finally (finally always calls `api.logout()`):
       a. Determine dinoBrainReferer:
          `language === 'en' ? 'https://dinorank.com/en/dinobrain/' : 'https://dinorank.com/dinobrain/'`
       b. `const brainHtml = await api.get(dinoBrainReferer, dinoBrainReferer)`
       c. `const credits = api.extractContentCredits(brainHtml)` — throw if 0

       d. Build context string from `WRITING_INSTRUCTIONS` + params.context + language/country rules
       e. Build generation body:
          ```typescript
          const genBody = new URLSearchParams({
            t: String(Date.now()),
            keyword: params.keyword,
            imagenes: 'no',
            contexto: contextString,
            exclusiones: params.exclusions || CONTENT_EXCLUSIONS,
            numPalabras: String(params.numWords || 2000),
            keyword_idioma: language === 'en' ? 'English' : 'Spanish',
            idioma: language,
          })
          if (params.country) {
            genBody.set('keyword_pais', params.country)
            genBody.set('pais', params.country)
          }
          ```
       f. POST `https://dinorank.com/ajax/generaContenido.php` with referer = dinoBrainReferer

       g. **Dual response parsing (CRITICAL — Pitfall 2):**
          ```typescript
          const trimmed = genRes.trim()
          let idContenido: string
          if (/^\d+$/.test(trimmed)) {
            idContenido = trimmed  // New API: plain integer
          } else {
            const genData = JSON.parse(trimmed)
            if (genData.status !== 'OK') throw new Error(`generaContenido error: ${genRes}`)
            idContenido = genData.message  // Legacy: JSON
          }
          ```

       h. **Poll controlIA.php** every `pollingDelay` ms (default 20000), max 40 attempts:
          ```typescript
          const pollBody = `t=${Date.now()}&idContenido=${idContenido}`
          // POST https://dinorank.com/ajax/controlIA.php
          // Check response for 'finalizado' or '100%'
          ```

       i. **Retrieve content:**
          POST `https://dinorank.com/ajax/obtieneContenidoGenerado.php`
          Body: `t=${Date.now()}&id=${idContenido}&modo=undefined&keyword=${encodeURIComponent(keyword)}`

       j. **extractBodyContent(html, keyword):**
          - `const { JSDOM } = await import('jsdom')` (dynamic import)
          - `const TurndownService = (await import('turndown')).default`
          - Parse HTML with JSDOM
          - Select `#textodelcontenido` or fall back to `doc.body`
          - Remove all `script`, `style`, `button`, `.btn`, `.toolbar` elements
          - Extract and remove `h1` → use as title (fallback: `Post sobre ${keyword}`)
          - `const td = new TurndownService({ headingStyle: 'atx', bulletListMarker: '-', codeBlockStyle: 'fenced' })`
          - `td.remove(['script', 'style', 'button'])`
          - `const markdown = td.turndown(container.innerHTML.trim())`
          - Clean HTML comments `<!--...-->` and 3+ consecutive blank lines

       k. **Mark cooldown:**
          ```typescript
          updateAccount(account.email, {
            cooldownUntil: new Date(Date.now() + 5 * 60 * 1000).toISOString(),
            contentCredits: Math.max(0, account.contentCredits - 1),
          })
          ```

       l. Return `{ title, markdown, accountEmail: account.email }`
