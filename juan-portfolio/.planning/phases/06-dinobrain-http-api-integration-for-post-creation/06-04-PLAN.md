    **3. Update registerAccount() signature — add optional profile param (backward-compatible):**
    ```typescript
    export function registerAccount(email: string, pass: string, profile?: AccountRegistrationProfile): AccountEntry {
      const now = new Date()
      const expires = new Date()
      expires.setDate(now.getDate() + 7)

      const newAcc: AccountEntry = {
        email,
        password: pass,
        kwCredits: 150,
        contentCredits: 5,
        keywords: [],
        content: [],
        createdAt: now.toISOString(),
        expiresAt: expires.toISOString(),
        lastUsed: now.toISOString(),
        // Profile fields — only set if profile provided
        ...(profile?.language && { createdLanguage: profile.language }),
        ...(profile?.country && { createdCountry: profile.country }),
        ...(profile?.domain && { createdDomain: profile.domain }),
        ...(profile?.projectType && { createdProjectType: profile.projectType }),
        ...(profile?.clientId && { assignedClientId: profile.clientId }),
      }

      const registry = loadRegistry()
      registry.push(newAcc)
      saveRegistry(registry)
      return newAcc
    }
    ```

    - `updateAccount()` — already uses `Partial<AccountEntry>` spread, so cooldownUntil works automatically
